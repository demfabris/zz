# set-hook -E fires a user event through the @ hook set-hook stored, with the
# target's session, window and pane as its payload, and wait-for -l and -w on a
# plain channel with no waiters list nothing and wake nobody.
set-hook -g @ping "set -gF @pong '#{hook}/#{hook_event}/#{hook_session_name}/#{hook_window_index}/#{hook_pane}'"
set-hook -E -t w:0 @ping
out: show -gv @pong
set-hook -E ping
set-hook -E
set-hook -E @a @b
set-hook -E @nohook
set-hook -gu @ping
out: wait-for -l hooks-events-chan
wait-for -w nobody hooks-events-chan
wait-for -S hooks-events-chan2
out: wait-for -l hooks-events-chan2
wait-for hooks-events-chan2
set -g @flag 0
set-hook -T -B '@mon::#{@flag}' "set -ga @monlog '#{hook_value}+'"
out: show-hooks -B
set-hook -u -B @mon
