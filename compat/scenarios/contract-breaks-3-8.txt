# fall back to an index or a name: +foo, -0 and +0 all fail for windows and
# panes, while a bare + or - and a valid offset still move.
new-window -d -t w -n two
new-window -d -t w -n three
split-window -d -t w:0
err: select-window -t w:+foo
err: select-window -t w:-0
err: select-window -t w:+0
err: select-window -t w:+x1
err: select-pane -t w:0.+foo
err: select-pane -t w:0.-0
err: select-pane -t w:0.+0
err: display-message -p -t w:+1 #{window_index}
err: display-message -p -t w:- #{window_index}
err: display-message -p -t w:0.+1 #{pane_index}
err: display-message -p -t w:0.- #{pane_index}
