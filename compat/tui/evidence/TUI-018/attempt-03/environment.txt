#!/usr/bin/env bash
# Two-slot, memory-capped cargo for the alienware box (15 GB RAM + zram).
# Every cargo command in cycle 11 goes through this. Never run bare cargo.
set -uo pipefail
CAP="${ZZ_CARGO_CAP:-5G}"
SWAP="${ZZ_CARGO_SWAP:-3G}"
JOBS="${ZZ_CARGO_JOBS:-3}"
S=$((RANDOM % 2))
exec systemd-run --user --scope -q \
  -p MemoryMax="$CAP" -p MemorySwapMax="$SWAP" \
  flock -w 900 /tmp/zz-cargo-slot-$S.lock \
  cargo "$@" --jobs "$JOBS"
xterm-ghostty
