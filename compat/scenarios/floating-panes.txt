out: display-message -p -t w #{window_layout}
split-window -d -h -t w.0
select-pane -t w.0
new-pane
out: list-panes -t w -F #{pane_id}:#{pane_width}x#{pane_height}:#{pane_x},#{pane_y}:#{pane_z}:#{pane_floating_flag}#{pane_active}#{pane_modal_flag}:#{pane_flags}

move-pane -L 2 -t w.1
out: list-panes -t w -F #{pane_id}:#{pane_width}x#{pane_height}:#{pane_x},#{pane_y}:#{pane_z}:#{pane_floating_flag}#{pane_active}#{pane_modal_flag}:#{pane_flags}
move-pane -z 1 -t w.1
out: list-panes -t w -F #{pane_id}:#{pane_width}x#{pane_height}:#{pane_x},#{pane_y}:#{pane_z}:#{pane_floating_flag}#{pane_active}#{pane_modal_flag}:#{pane_flags}
out: display-message -p -t w #{window_layout}
move-pane -P front -t w.2

move-pane -X 5 -Y 4 -t w.3

join-pane -s w.1 -t w.1
out: list-panes -t w -F #{pane_id}:#{pane_width}x#{pane_height}:#{pane_x},#{pane_y}:#{pane_z}:#{pane_floating_flag}#{pane_active}#{pane_modal_flag}:#{pane_flags}

out: display-message -p -t w #{window_layout}
resize-window -t w -x 30 -y 10

out: display-message -p -t w #{window_layout}

# Errors from the floating commands.
new-session -d -s errs
new-pane -t errs
err: break-pane -W -s errs.1
err: move-pane -P top-left -t errs.0
err: move-pane -P sideways -t errs.1
err: join-pane -s errs.0 -t errs.1
err: split-window -L -t errs.1
err: new-pane -L -t errs.1
err: new-pane -L -O -t errs.0
err: break-pane -W -s errs.0 -x 0
err: join-pane -s errs.0 -t errs.0
new-pane -O -t errs.0
err: new-pane -O -t errs.0
err: break-pane -s errs.2
err: join-pane -s errs.2 -t errs.0
out: list-panes -t errs -F #{pane_id}:#{pane_width}x#{pane_height}:#{pane_x},#{pane_y}:#{pane_z}:#{pane_floating_flag}#{pane_active}#{pane_modal_flag}:#{pane_flags}
out: display-message -p -t errs #{window_modal_pane}:#{window_flags}
