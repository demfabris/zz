# Every after-* and event hook compat/census.py reported as registered but
# never triggered by any proof, fired here through set-hook writing a global
# environment variable and read back with show-environment. Each name is
# pre-seeded to no so the readback compares a value, not an exit class, and
# every resize is restored so a later window inherits the same size on both.
# Eight of the 37 the census listed cannot be fired detached and are recorded
# with the reason under semantic:census-hooks in proofs.census-coverage:
# after-copy-mode, after-display-panes, after-refresh-client, alert-activity,
# alert-silence, client-dark-theme, client-light-theme and window-unlinked.
set-environment -g CENSUS_AFTER_KILL_PANE no
set-hook -g after-kill-pane "set-environment -g CENSUS_AFTER_KILL_PANE yes"
split-window -t w:0
kill-pane -t w:0.1
set-hook -gu after-kill-pane
out: show-environment -g CENSUS_AFTER_KILL_PANE
