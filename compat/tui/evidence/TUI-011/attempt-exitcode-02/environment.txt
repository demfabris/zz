# Two-slot, memory-capped cargo for the alienware box (15 GB RAM + zram).
# Every cargo command in cycle 11 goes through this. Never run bare cargo.

SWAP="${ZZ_CARGO_SWAP:-2G}"

#!/usr/bin/env python3
import os,pathlib,subprocess,sys,tempfile,time
with tempfile.TemporaryDirectory(prefix='zz-exit-cargo-') as directory:
    marker=pathlib.Path(directory)/'started'
    env={**os.environ,'PATH':'/tmp/zz-exitcode-cargo/inside:'+os.environ['PATH'],'ZZ_EXITCODE_CARGO_STARTED':str(marker)}
    attempt=0
    while True:
        attempt+=1
        child=subprocess.Popen(['/tmp/zz-cargo.sh',*sys.argv[1:]],env=env)
        deadline=time.monotonic()+15
        while child.poll() is None and not marker.exists() and time.monotonic()<deadline:
            time.sleep(.05)
        if marker.exists() or child.poll() is not None:
            sys.exit(child.wait())
        try:
            unit=pathlib.Path(f'/proc/{child.pid}/cgroup').read_text().strip().split('/')[-1]
            children=pathlib.Path(f'/proc/{child.pid}/task/{child.pid}/children').read_text().strip()
            if unit.startswith('run-') and unit.endswith('.scope') and not marker.exists() and not children:
                print(f'exitcode queue adapter: retry {attempt}; no cargo start marker after 15 seconds; terminating owned queued scope {unit}',file=sys.stderr,flush=True)
                subprocess.run(['systemctl','--user','kill',unit],check=True)
                status=child.wait()
                if marker.exists():
                    sys.exit(status)
            else:
                sys.exit(child.wait())
        except FileNotFoundError:
            sys.exit(child.wait())
