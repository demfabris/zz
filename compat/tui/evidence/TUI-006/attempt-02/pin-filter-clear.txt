pin-only probe, choose-tree -Zw: open, f, a window filter, Enter, c
script:
  #!/usr/bin/env bash
  set -eEuo pipefail
  TMUXB=/home/demfabris/dev/zz/compat/.cache/tmux-src/tmux
  D=/tmp/zzprobe-filter.$$
  rm -rf "$D"; mkdir -p "$D/inner" "$D/outer"
  OUT="zzprobe-o-$$"; INN="zzprobe-i-$$"
  SH="ENV= PS1='\$ ' exec /bin/sh"
  inner() { env -u TMUX -u TMUX_PANE TMUX_TMPDIR=/tmp HOME="$D/inner" XDG_CONFIG_HOME="$D/inner/c" "$TMUXB" -L "$INN" "$@"; }
  outer() { env -u TMUX -u TMUX_PANE TMUX_TMPDIR=/tmp HOME="$D/outer" XDG_CONFIG_HOME="$D/outer/c" "$TMUXB" -L "$OUT" "$@"; }
  cleanup() { outer kill-server >/dev/null 2>&1 || true; inner kill-server >/dev/null 2>&1 || true; rm -rf "$D"; }
  trap cleanup EXIT INT TERM
  inner -f /dev/null new-session -d -s cho -n win -x 80 -y 24 "$SH"
  inner set-option -g status-right ''
  inner set-option -g status-left L
  inner set-option -g automatic-rename off
  inner new-window -d -t '=cho:1' -n two "$SH"
  inner split-window -d -h -t '=cho:1' "$SH"
  inner new-session -d -s many -n m00 -x 80 -y 24 "$SH"
  inner new-window -d -t '=many:1' -n m01 "$SH"
  inner new-window -d -t '=many:2' -n m02 "$SH"
  inner new-session -d -s other -n far -x 80 -y 24 "$SH"
  for p in $(inner list-panes -a -F '#{pane_id}'); do inner select-pane -t "$p" -T ptitle; done
  printf '#!/usr/bin/env bash\nexec env -u TMUX -u TMUX_PANE TMUX_TMPDIR=/tmp HOME=%q XDG_CONFIG_HOME=%q %q -L %q attach-session -t =cho\n' \
    "$D/inner" "$D/inner/c" "$TMUXB" "$INN" > "$D/att.sh"
  chmod +x "$D/att.sh"
  outer -f /dev/null new-session -d -s drv -n p -x 80 -y 24 "$D/att.sh"
  outer set-option -g status off
  sleep 1.5
  snap() { printf '=== %s ===\n' "$1"; outer capture-pane -p -t '=drv:p'; }
  key() { outer send-keys -t '=drv:p' "$@"; sleep 1.2; }
  key C-b; key w
  snap open
  key f
  snap prompt
  key -l '#{==:#{window_name},two}'
  key Enter
  snap applied
  key c
  snap cleared
