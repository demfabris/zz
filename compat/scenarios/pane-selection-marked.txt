# One server-global mark (server.c server_set_marked, cmd-select-pane.c -m/-M).
# A second -m moves it, re-marking the holder clears it through
# server_is_marked, -M clears whatever holds it, and the marked formats read
# per pane, window, and session.
split-window -d -t w:0.0
out: display-message -p -t w:0.0 #{pane_marked}:#{pane_marked_set}
out: display-message -p -t w:0 #{window_marked_flag}:[#{window_flags}]
out: display-message -p -t w #{session_marked}
select-pane -m -t w:0.0
out: display-message -p -t w:0.0 #{pane_marked}:#{pane_marked_set}
out: display-message -p -t w:0.1 #{pane_marked}:#{pane_marked_set}
out: display-message -p -t w:0 #{window_marked_flag}:[#{window_flags}]
out: display-message -p -t w #{session_marked}
select-pane -m -t w:0.1
out: display-message -p -t w:0.0 #{pane_marked}:#{pane_marked_set}
out: display-message -p -t w:0.1 #{pane_marked}:#{pane_marked_set}
select-pane -m -t w:0.1
out: display-message -p -t w:0.1 #{pane_marked}:#{pane_marked_set}
out: display-message -p -t w:0 [#{window_flags}]
select-pane -m -t w:0.0
select-pane -M -t w:0.1
out: display-message -p -t w:0.0 #{pane_marked}:#{pane_marked_set}
select-pane -M -t w:0.0
out: display-message -p -t w:0.0 #{pane_marked_set}

# The mark scopes to the session and window that own it, while pane_marked_set
# answers server-wide.
new-window -d -t w -n second
new-session -d -s other
select-pane -m -t w:0.0
out: display-message -p -t w #{session_marked}
out: display-message -p -t other #{session_marked}
out: display-message -p -t w:0 #{window_marked_flag}:[#{window_flags}]
out: display-message -p -t w:second #{window_marked_flag}:[#{window_flags}]
out: display-message -p -t other:0 #{window_marked_flag}
out: display-message -p -t w:second.0 #{pane_marked_set}

# window_lost_pane clears the mark when the marked pane dies.
kill-pane -t w:0.0
out: display-message -p -t w:0.0 #{pane_marked}:#{pane_marked_set}
out: display-message -p -t w #{session_marked}
out: display-message -p -t w:0 [#{window_flags}]

# -m refuses a pane hidden by zoom (window_pane_is_visible) but -M still clears.
split-window -d -t w:0.0
resize-pane -Z -t w:0.0
select-pane -m -t w:0.1
out: display-message -p -t w:0.0 #{pane_marked_set}
select-pane -m -t w:0.0
out: display-message -p -t w:0.0 #{pane_marked}:#{pane_marked_set}
out: display-message -p -t w:0 [#{window_flags}]
select-pane -M -t w:0.1
out: display-message -p -t w:0.0 #{pane_marked_set}

# cmd_find_valid_state ends at window_has_pane, so moving the marked pane into
# another window through break-pane or join-pane invalidates the mark while a
# swap inside the same window keeps it.
select-pane -M -t w:0.0
new-window -d -t w:8 -n sidecar
split-window -d -t w:0.0
select-pane -m -t w:0.1
out: display-message -p -t w:0.0 #{pane_marked_set}
break-pane -d -s w:0.1 -t w:9 -n moved
out: display-message -p -t w:0.0 #{pane_marked_set}
select-pane -m -t w:moved.0
out: display-message -p -t w:0.0 #{pane_marked_set}
join-pane -d -s w:moved.0 -t w:sidecar.0
out: display-message -p -t w:0.0 #{pane_marked_set}
select-pane -m -t w:sidecar.1
out: display-message -p -t {marked} #{window_name}.#{pane_index}
swap-pane -d -s w:sidecar.1 -t w:sidecar.0
out: display-message -p -t w:0.0 #{pane_marked_set}
out: display-message -p -t {marked} #{window_name}.#{pane_index}
select-pane -M -t w:0.0

# marked_pane.s is never retargeted: server_link_window and cmd_swap_window_exec
# only retarget marked_pane.wl, and cmd_find_valid_state walks that session's
# winlinks, so a move inside the session keeps the mark while a window that
# changes session takes no mark with it.
select-pane -m -t w:sidecar.1
move-window -d -s w:sidecar -t w:7
out: display-message -p -t w:7.1 #{pane_marked}:#{pane_marked_set}:#{session_marked}:#{window_marked_flag}
out: display-message -p -t w:7 [#{window_flags}]
new-window -d -t other:1 -n spare
move-window -d -s w:7 -t other:5
out: display-message -p -t other:5.1 #{pane_marked}:#{pane_marked_set}:#{session_marked}:#{window_marked_flag}
out: display-message -p -t other:5 [#{window_flags}]
out: display-message -p -t w #{session_marked}
select-pane -m -t w:second.0
swap-window -d -s w:second -t other:1
out: display-message -p -t other:1.0 #{pane_marked}:#{pane_marked_set}:#{session_marked}:#{window_marked_flag}
out: display-message -p -t other:1 [#{window_flags}]
out: display-message -p -t w:1.0 #{pane_marked}:#{pane_marked_set}
